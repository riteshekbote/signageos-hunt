            url: apiUrl,
            version: 'v1',
            auth,

        const nowDate = new Date();

                appletUid,

                    sync_server: syncServer,
                    sync_group: syncGroup,
                    video_uri: video.uri,
                },
                startsAt: nowDate,
                endsAt: nowDate,
                finishEvent: {
                    type: 'DURATION',
                    data: null,
                },
                position: 1,
            })
        }
    },
